\begin{figure}[h]
\centering
\begin{tikzpicture}
   % Clase base
   \umlclass{Vehiculo}{
      % Atributos
      - matricula : string \\
      - velocidad : float
   }{
      + acelerar(v : float) : void \\
      + frenar() : void \\
      \# revisar() : bool
   }

   \umlclass[y=-4]{Coche}{
      - puertas : int
   }{
      + abrirMaletero() : void
   }

   \umlclass[x=6]{Garaje}{
      plazas : int \\ direccion : string \\ abierto : bool
   }{}

   \umlclass[x=6, y=-4]{Conductor}{nombre : string}{+ conducir() : void}

   \umlrelationship[type=inheritance]{Coche}{Vehiculo}
   \umlrelationship{Garaje}{Vehiculo} % sin tipo: asociación
\end{tikzpicture}

\end{figure}
